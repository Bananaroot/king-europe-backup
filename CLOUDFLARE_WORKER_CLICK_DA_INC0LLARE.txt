/* KING BENESSERE CLICK - Worker separato, solo conteggio clic eBay.
   Richiede GITHUB_TOKEN come segreto di Cloudflare, accesso solo
   a Bananaroot/king-test-benessere-dati (Contents read/write).
   GET /go?u=LINK_EBAY&n=PRODOTTO&v=v2-3.0&c=CAMPAGNA
*/
const REPO = 'Bananaroot/king-test-benessere-dati';
const FILE = 'clic-prodotti.json';
const API = `https://api.github.com/repos/${REPO}/contents/${FILE}`;
function decode(content){
  return JSON.parse(new TextDecoder().decode(Uint8Array.from(atob(content.replace(/\s/g,'')),x=>x.charCodeAt(0))));
}
function encode(text){
  const bytes=new TextEncoder().encode(text);let b='';
  for(let i=0;i<bytes.length;i++)b+=String.fromCharCode(bytes[i]);
  return btoa(b);
}
function isEbay(url){
  try{
    const u=new URL(url);const h=u.hostname.toLowerCase();
    return u.protocol==='https:'&&(h==='ebay.io'||h==='www.ebay.io'||/(^|\.)ebay\.(it|com|co\.uk|de|fr|es|pl)$/.test(h));
  }catch(_){return false}
}
function headers(token){return {
  Authorization:`Bearer ${token}`,'Accept':'application/vnd.github+json',
  'X-GitHub-Api-Version':'2022-11-28','Content-Type':'application/json',
  'User-Agent':'king-benessere-click'
}}
async function register(env,item){
  if(!env.GITHUB_TOKEN)return false;
  const head=headers(env.GITHUB_TOKEN);
  for(let attempt=0;attempt<4;attempt++){
    const r=await fetch(API+'?ref=main',{headers:head,cache:'no-store'});
    let sha,db={totale:0,links:[],per_giorno:{},per_campagna:{},ultimo_aggiornamento:null};
    if(r.status===200){const f=await r.json();sha=f.sha;db=decode(f.content)}
    else if(r.status!==404)throw Error('GitHub read '+r.status);
    if(!Array.isArray(db.links))db.links=[];
    const k=item.url;let rec=db.links.find(x=>x.url===k);
    if(!rec){rec={url:item.url,nome:item.name,clic:0};db.links.push(rec)}
    rec.clic=Number(rec.clic||0)+1;
    rec.nome=item.name;
    db.totale=Number(db.totale||0)+1;
    const day=new Date().toISOString().slice(0,10);
    db.per_giorno??={};db.per_giorno[day]=Number(db.per_giorno[day]||0)+1;
    db.per_campagna??={};if(item.campaign)db.per_campagna[item.campaign]=Number(db.per_campagna[item.campaign]||0)+1;
    db.ultimo_aggiornamento=new Date().toISOString();
    const data={message:'KING Click - '+day,branch:'main',content:encode(JSON.stringify(db))};
    if(sha)data.sha=sha;
    const w=await fetch(API,{method:'PUT',headers:head,body:JSON.stringify(data)});
    if(w.ok)return true;
    if(w.status!==409&&w.status!==422)throw Error('GitHub write '+w.status);
  }
  return false;
}
export default {
  async fetch(request,env){
    const u=new URL(request.url);
    if(u.pathname==='/health')return Response.json({ok:true,service:'KING BENESSERE CLICK'});
    if(request.method!=='GET'||u.pathname!=='/go')return new Response('Not found',{status:404});
    const dest=u.searchParams.get('u')||'';
    if(!isEbay(dest))return new Response('Link eBay non valido',{status:400});
    const name=(u.searchParams.get('n')||'Prodotto').slice(0,120);
    const campaign=(u.searchParams.get('c')||'').replace(/[^a-zA-Z0-9_-]/g,'').slice(0,60);
    try{await register(env,{url:dest,name,campaign})}
    catch(e){console.error('CLICK_NOT_REGISTERED',String(e))}
    // Anche se GitHub non risponde, il cliente arriva al prodotto.
    return Response.redirect(dest,302);
  }
};
